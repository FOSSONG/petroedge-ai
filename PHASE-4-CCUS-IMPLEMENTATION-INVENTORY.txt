56:     sample: WellLogSample,
57: ) -> float:
58:     """
59:     Deterministic operational fallback used only when a registered model
60:     is unavailable. It preserves service availability while making the
61:     fallback explicit in the explanation payload.

63:     resistivity_score = min(
64:         max(sample.resistivity_ohmm / 100.0, 0.0),
65:         1.0,

77:         max(sample.neutron_porosity_vv / 0.30, 0.0),
78:         1.0,
79:     )

82:         0.50 * resistivity_score
83:         + 0.30 * clean_sand_score
84:         + 0.20 * porosity_score
85:     )

88:         min(max(probability, 0.0), 1.0),
